interface Printable extends \Stringable
{
    public const FORMAT = "%-20s %10.2f";
    public function render(bool $html = false): string;
}

trait Auditable
{
    private array $log = [];
    protected static int $instances = 0;

    public function audit(string $event, mixed ...$context): static
    {
        $this->log[] = [$event, $context, time()];
        return $this;
    }
    abstract public function id(): string|int;
}

enum Status: string
{
    case Draft = 'draft';
    case Sent = 'sent';
    case Paid = 'paid';

    public function label(): string
    {
        return match ($this) {
            self::Draft => 'Taslak',
            self::Sent, self::Paid => ucfirst($this->value),
        };
    }
}

#[Route('/invoices/{id}', methods: ['GET', 'POST'])]
class Invoice implements Printable, \Countable
{
    use Auditable;

    public const TAX = 0.18;
    private const MASK = 0xFF_EC, PERMS = 0o644, BITS = 0b1010_0101, BIG = 1_000_000, TINY = 1.5e-3, OLD = 0755;
    public ?Customer $customer = null;
    protected array $lines = [];
    private static ?self $last = null;

    public function __construct(
        private readonly string $number,
        protected Status $status = Status::Draft,
        public readonly DateTimeImmutable $issued = new DateTimeImmutable('now'),
    ) {
        static::$instances++;
        self::$last = $this;
    }

    public function __get(string $name): mixed
    {
        return $name === 'total' ? $this->total() : throw new BadInput("No property $name");
    }

    public function id(): string { return $this->number; }
    public function count(): int { return \count($this->lines); }

    public function add(string $label, float|int $price, int $qty = 1): self
    {
        if ($qty <= 0 or $price < 0) {
            throw new BadInput(sprintf('Bad line "%s": qty=%d', $label, $qty));
        }
        $this->lines[] = compact('label', 'price', 'qty');
        return $this->audit('line.added', label: $label);
    }

    public function total(bool $withTax = true): float
    {
        $sum = array_sum(array_map(fn(array $l): float => $l['price'] * $l['qty'], $this->lines));
        return round($withTax ? $sum * (1 + self::TAX) : $sum, 2);
    }

    public function render(bool $html = false): string
    {
        $city = $this->customer?->address?->city ?? 'unknown';
        $rows = '';
        foreach ($this->lines as $i => ['label' => $label, 'price' => $price, 'qty' => $qty]) {
            $rows .= sprintf(static::FORMAT, "$qty × {$label}", (float) $price * (int) $qty) . PHP_EOL;
        }
        $first = $this->lines[0] ?? ['label' => 'n/a'];
        $summary = "Invoice #{$this->number} for $city: {$first['label']} and ${rows}\t\x41\u{1F9FE}\n";
        $text = <<<EOT
            Invoice {$this->number} ({$this->status->label()})
            Issued: {$this->issued->format('Y-m-d')}  Lines: {$this->count()}
            $rows
            Total: {$this->total()} ₺ — "quoted" and \$escaped
            EOT;
        $legal = <<<'EOT'
            Nothing is interpolated here: $rows {$this->number} \n stays.
            EOT;
        return $html ? '<pre>' . htmlspecialchars($text . $legal, ENT_QUOTES) . '</pre>' : $text . $summary;
    }

    public function __toString(): string { return $this->render(); }
    public static function last(): ?static { return static::$last; }
    public function __destruct() { /* nothing to release */ }
}

function discount(Invoice $invoice, callable|null $rule = null): \Closure
{
    $rate = 0.1;
    $rule ??= function (float $total) use ($rate, &$invoice): float {
        return $total > 1_000 ? $total * (1 - $rate) : $total;
    };
    return static fn(): float => $rule($invoice->total());
}
