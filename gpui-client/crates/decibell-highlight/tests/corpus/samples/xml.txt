<project name="reading-room" default="dist" basedir=".">
  <property name="src" location="src"/>
  <property name="build" location="${basedir}/build"/>
  <property environment="env"/>

  <path id="classpath">
    <fileset dir="lib" includes="**/*.jar" excludes="**/*-sources.jar"/>
    <pathelement path="${env.JAVA_HOME}/lib/tools.jar"/>
  </path>

  <target name="init" description="Create the build directory">
    <tstamp><format property="today" pattern="yyyy-MM-dd"/></tstamp>
    <mkdir dir="${build}"/>
  </target>

  <target name="compile" depends="init" unless="skip.compile">
    <javac srcdir="${src}" destdir="${build}" includeantruntime="false" debug="on" source="21">
      <classpath refid="classpath"/>
      <compilerarg value="-Xlint:all"/>
    </javac>
    <echo message="Compiled at ${today}: &quot;done&quot;" level="info"/>
  </target>

  <target name="dist" depends="compile">
    <jar jarfile="dist/reading-room-${today}.jar" basedir="${build}">
      <manifest>
        <attribute name="Main-Class" value="org.example.Main"/>
        <attribute name="Implementation-Title" value='Reading "Room"'/>
      </manifest>
    </jar>
  </target>
</project>
