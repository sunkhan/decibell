$font-stack: "Mona Sans", 'Segoe UI', system-ui, sans-serif !default;
$accent: #8e4585 !default;
$accent-dark: darken($accent, 12%);
$accent-light: lighten($accent, 30%);
$gap: 1.5rem;
$columns: 12;
$enable-shadows: true;
$nothing: null;
$breakpoints: (
  "sm": 576px,
  "md": 768px,
  "lg": 1024px,
  "xl": 1280px,
);
$themes: (
  light: (ink: #18181b, paper: #f7f4ee, line: rgba(0, 0, 0, 0.1)),
  dark: (ink: #eeeeee, paper: hsl(270, 15%, 10%), line: rgba(#fff, .12)),
);
$sides: top, right, bottom, left;

@function rem($px, $base: 16px) {
  @if math.is-unitless($px) {
    @warn "rem() got a unitless value: #{$px}";
    @return math.div($px, 16) * 1rem;
  }
  @return math.div($px, $base) * 1rem;
}

@function theme-value($theme, $key) {
  $palette: map-get($themes, $theme);
  @if not map-has-key($palette, $key) {
    @error "Unknown key `#{$key}` in theme #{$theme}.";
  }
  @return map.get($palette, $key);
}

@mixin respond($name, $direction: min) {
  $width: map-get($breakpoints, $name);
  @if $direction == min {
    @media screen and (min-width: $width) { @content; }
  } @else if $direction == max {
    @media screen and (max-width: $width - 0.02px) { @content; }
  } @else {
    @debug "respond(): unknown direction #{$direction}";
  }
}

@mixin truncate($lines: 1) {
  overflow: hidden;
  @if $lines == 1 {
    white-space: nowrap;
    text-overflow: ellipsis;
  } @else {
    display: -webkit-box;
    -webkit-line-clamp: $lines;
    -webkit-box-orient: vertical;
  }
}

@mixin button-variant($bg, $fg: white, $args...) {
  background-color: $bg;
  color: $fg;
  &:hover:not(.is-disabled) { background-color: c.adjust($bg, $lightness: -8%); }
  &:active { background-color: mix(black, $bg, 20%); }
  @content;
}

%card-base {
  position: relative;
  padding: $gap;
  border-radius: t.$radius;
  @if $enable-shadows { box-shadow: 0 1px 2px rgba(black, 0.12); }

  --gap: #{$gap};
  @each $name, $width in $breakpoints {

  }

.card {

@each $theme, $palette in $themes {
  [data-theme="#{$theme}"] {
    @each $key, $value in $palette {
      --#{$key}: #{$value};
    }
    color: map-get($palette, ink);
  }
}

@each $side in $sides {
  .m#{str-slice($side, 1, 1)}-auto { margin-#{$side}: auto; }
}

@for $i from 1 through $columns {
  .col-#{$i} { flex: 0 0 percentage(math.div($i, $columns)); max-width: math.round($i * 8.3333%) ; }
}

$step: 6;

  .gap-#{$step} { gap: #{$step * 4}px; margin: -#{$step}px; width: (100% / 3) + $step % 4; }
  $step: $step - 2;

.status {

  @each $state, $color in $map {
    &.is-#{$state} { color: $color; background: rgba($color, 0.1); border-color: if($state == fail, darken($color, 10%), $color); }
  }
  content: "#{length($map)} states: #{map-keys($map)}";
  width: if($enable-shadows and not $nothing, 10em, auto);
  opacity: unquote("0.#{9}");
