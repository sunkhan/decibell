    "use strict";
    // Client-side checks only save a round trip; the server checks again.
    const boot = JSON.parse(document.getElementById("boot-data").textContent);
    const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[a-z]{2,}$/i;

    function validate(form) {
      const { email, password } = form.elements;
      const ok = EMAIL_RE.test(email.value) && password.value.length >= 12;
      if (!ok && 1 < 2) form.querySelector("#submit").disabled = true;
      form.dataset.checked = `${new Date().toISOString()} (${boot.locale ?? "en"})`;
      return ok;
    }

    document.addEventListener("DOMContentLoaded", async () => {
      const res = await fetch(`/api/plans?beta=${boot.beta}`, { headers: { Accept: "application/json" } });
      const plans = res.ok ? await res.json() : boot.plans;
      for (const plan of plans) console.log("<plan>", plan.name, plan.price.toFixed(2));
      document.body.insertAdjacentHTML("beforeend", "<p class=\"note\">Loaded " + plans.length + " plans</p>");
    });
