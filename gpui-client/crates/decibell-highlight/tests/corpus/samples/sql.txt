BEGIN TRANSACTION;

CREATE SCHEMA IF NOT EXISTS shop AUTHORIZATION shop_owner;
SET search_path TO shop, public;

CREATE TYPE order_status AS ENUM ('new', 'paid', 'shipped', 'cancelled');

CREATE TABLE IF NOT EXISTS authors (
    author_id   BIGSERIAL PRIMARY KEY,
    full_name   VARCHAR(120) NOT NULL,
    born        DATE,
    bio         TEXT DEFAULT '',
    "Sort Key"  CHARACTER VARYING(120) GENERATED ALWAYS AS (lower(full_name)) STORED,
    CONSTRAINT authors_name_unique UNIQUE (full_name)
);

CREATE TABLE books (
    book_id     INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    author_id   BIGINT NOT NULL REFERENCES authors (author_id) ON DELETE CASCADE ON UPDATE NO ACTION,
    isbn        CHAR(13) UNIQUE,
    title       TEXT NOT NULL CHECK (char_length(title) > 0),
    price       NUMERIC(10, 2) NOT NULL DEFAULT 9.99,
    weight_kg   DOUBLE PRECISION DEFAULT 2.5e-1,
    in_print    BOOLEAN NOT NULL DEFAULT TRUE,
    tags        TEXT[] DEFAULT ARRAY['new']::text[],
    metadata    JSONB,
    added_at    TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (author_id) REFERENCES authors (author_id)
);

create table orders (
    order_id    bigint primary key,
    book_id     int not null references books,
    quantity    smallint not null default 1 check (quantity between 1 and 1000),
    status      order_status not null default 'new',
    placed_at   timestamptz not null default now(),
    note        text null
);

CREATE UNIQUE INDEX CONCURRENTLY idx_books_isbn ON books (isbn) WHERE isbn IS NOT NULL;
Create Index idx_orders_placed On orders Using btree (placed_at Desc, status);

CREATE OR REPLACE VIEW bestsellers AS
SELECT b.book_id, b.title, SUM(o.quantity) AS sold
FROM books AS b
INNER JOIN orders o ON o.book_id = b.book_id
WHERE o.status <> 'cancelled'
GROUP BY b.book_id, b.title
HAVING SUM(o.quantity) >= 10
ORDER BY sold DESC NULLS LAST
LIMIT 20 OFFSET 0;

UPDATE books SET price = price * 1.10, in_print = FALSE
WHERE added_at < NOW() - INTERVAL '5 years' AND title NOT LIKE '%Collected%' ESCAPE '\';

DELETE FROM orders WHERE status = 'cancelled' AND placed_at < CURRENT_DATE - 365;

-- A recursive CTE: number the days of one month.
WITH RECURSIVE days(day) AS (
    SELECT DATE '2026-01-01'
    UNION ALL
    SELECT day + 1 FROM days WHERE day < DATE '2026-01-31'
),
daily AS (
    SELECT d.day, COALESCE(SUM(o.quantity * b.price), 0)::numeric(12, 2) AS revenue, COUNT(DISTINCT o.order_id) AS orders
    FROM days d
    LEFT OUTER JOIN orders o ON o.placed_at::date = d.day
    LEFT JOIN books b USING (book_id)
    GROUP BY d.day
)
SELECT day,
       revenue,
       SUM(revenue) OVER (ORDER BY day ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total,
       AVG(revenue) OVER w AS weekly_average,
       RANK() OVER (PARTITION BY EXTRACT(WEEK FROM day) ORDER BY revenue DESC) AS rank_in_week,
       LAG(revenue, 1, 0.0) OVER (ORDER BY day) AS previous_day,
       CASE
           WHEN revenue = 0 THEN 'none'
           WHEN revenue BETWEEN 0.01 AND 99.99 THEN 'slow'
           WHEN revenue >= 1e3 THEN 'great'
           ELSE 'fine'
       END AS verdict
FROM daily
WINDOW w AS (ORDER BY day ROWS 6 PRECEDING)
ORDER BY day ASC;

Select a.full_name || ' (' || Cast(Count(*) As varchar) || ')' As label,
       Upper(Substring(a.full_name From 1 For 1)) As initial,
       Round(Avg(b.price), 2) As avg_price, Max(b.price) - Min(b.price) As spread,
       b.metadata ->> 'publisher' As publisher, b.tags @> ARRAY['signed'] As signed
From authors a
Full Outer Join books b On b.author_id = a.author_id
Where b.price Is Not Null
  And a.author_id In (Select author_id From books Where in_print Is True Group By author_id Having Count(*) > 1)
  And Not Exists (Select 1 From orders o Where o.book_id = b.book_id And o.status = Any ('{cancelled}'))
  And (a.born Is Null Or a.born >= $1) And a.full_name Ilike :pattern And b.price % 2 != 0
Group By a.author_id, a.full_name, b.metadata, b.tags
Union All
Select 'nobody', '?', 0, -1.5, Null, False
Except
Select 'x', 'x', 0x1F, 0, Null, Null;

CREATE FUNCTION restock(p_book INTEGER, p_amount INTEGER DEFAULT 10) RETURNS VOID AS $body$
DECLARE
    v_title TEXT;
BEGIN
    SELECT title INTO v_title FROM books WHERE book_id = p_book FOR UPDATE;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'book % not found', p_book;
    END IF;
    RAISE NOTICE 'restocking "%" by %', v_title, p_amount;
END;
$body$ LANGUAGE plpgsql;

# This line is a comment in MySQL only.
SET @min_price := 12.5, @label = "double-quoted string";
SELECT `b`.`title`, `b`.`price`, IFNULL(`b`.`isbn`, 'n/a') AS `isbn`, CONCAT_WS(' / ', `title`, @label)
FROM `shop`.`books` AS `b`
WHERE `b`.`price` >= @min_price AND `b`.`title` REGEXP '^[A-M]'
ORDER BY `b`.`price` DESC LIMIT 5, 10;

-- SQL Server flavour: brackets, TOP, variables and N'' strings.
DECLARE @cutoff DATETIME2 = DATEADD(day, -30, GETDATE());
SELECT TOP (10) [o].[order_id], [o].[quantity], N'Ünïcode' AS [Label Text], ISNULL([o].[note], '') AS note
FROM [shop].[dbo].[orders] AS [o] WITH (NOLOCK)
CROSS APPLY (SELECT COUNT(*) AS n FROM [books]) AS c
WHERE [o].[placed_at] >= @cutoff AND [o].[quantity] <= 0xFF
ORDER BY [o].[placed_at] DESC;
